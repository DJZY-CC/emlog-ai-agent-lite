## M01：ReAct Agent 循环引擎

**适用系统**: emlog pro  
**适用场景**: 需要 AI 自主决策并调用多个工具完成复杂任务的插件。  
**依赖说明**: PHP 7.4+、cURL、OpenAI 兼容 API（支持 Function Calling）

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

class AIAgent_AgentLoop
{
    private $config;
    private $promptBuilder;
    private $toolRunner;
    private $maxRounds;
    private $log = [];

    public function __construct($config, $uid = 0, $role = 'guest')
    {
        $this->config = $config;
        $this->maxRounds = min(3, max(1, intval($config['agent_max_rounds'])));
        $this->promptBuilder = new AIAgent_PromptBuilder($config, $uid);
        $this->toolRunner = new AIAgent_ToolRunner($config, $uid, $role);
    }

    public function execute($user_message, $conversation_history = [])
    {
        $this->log = [];
        $tool_calls_log = [];
        $system_prompt = $this->promptBuilder->buildSystemPrompt();
        $tools_definition = $this->promptBuilder->getToolsDefinition();
        $messages = [['role' => 'system', 'content' => $system_prompt]];
        $history = array_slice($conversation_history, -20);
        foreach ($history as $msg) $messages[] = $msg;
        $messages[] = ['role' => 'user', 'content' => $user_message];

        $final_response = '';
        for ($round = 1; $round <= $this->maxRounds; $round++) {
            $llm_response = $this->callLLM($messages, $tools_definition);
            if (!$llm_response['success']) {
                return ['response' => 'AI服务不可用: ' . $llm_response['error'],
                    'log' => $this->log, 'tool_calls' => $tool_calls_log];
            }
            $assistant_message = $llm_response['data']['message'] ?? [];
            $tool_calls = $assistant_message['tool_calls'] ?? [];
            if (empty($tool_calls)) {
                $final_response = $assistant_message['content'] ?? '';
                break;
            }
            $messages[] = $assistant_message;
            foreach ($tool_calls as $tc) {
                $func = $tc['function'] ?? [];
                $name = $func['name'] ?? '';
                $args = is_string($func['arguments'] ?? '')
                    ? json_decode($func['arguments'], true) ?: []
                    : ($func['arguments'] ?? []);
                $result = $this->toolRunner->run($name, $args);
                $tool_calls_log[] = ['tool' => $name, 'args' => $args, 'result' => $result];
                $messages[] = [
                    'role' => 'tool', 'tool_call_id' => $tc['id'] ?? '',
                    'content' => json_encode($result, JSON_UNESCAPED_UNICODE),
                ];
            }
            if ($round === $this->maxRounds) {
                $summary = $this->callLLM($messages, []);
                $final_response = $summary['success']
                    ? ($summary['data']['message']['content'] ?? '操作已完成')
                    : '操作已完成，但摘要生成失败';
            }
        }
        return ['response' => $final_response, 'log' => $this->log, 'tool_calls' => $tool_calls_log];
    }

    private function callLLM($messages, $tools = [])
    {
        $post_data = [
            'model' => $this->config['llm_model'],
            'messages' => $messages,
            'temperature' => floatval($this->config['llm_temperature']),
            'max_tokens' => intval($this->config['llm_max_tokens']),
        ];
        if (!empty($tools)) {
            $formatted = [];
            foreach ($tools as $t) {
                $formatted[] = ['type' => 'function', 'function' => $t];
            }
            $post_data['tools'] = $formatted;
        }
        $json = json_encode($post_data, JSON_UNESCAPED_UNICODE);
        if ($json === false) {
            $post_data = $this->cleanUtf8($post_data);
            $json = json_encode($post_data, JSON_UNESCAPED_UNICODE);
        }
        $ch = curl_init($this->config['llm_api_url']);
        curl_setopt_array($ch, [
            CURLOPT_POST => true, CURLOPT_POSTFIELDS => $json,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json',
                'Authorization: Bearer ' . $this->config['llm_api_key']],
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => max(10, intval($this->config['llm_timeout'] ?? 60)),
            CURLOPT_SSL_VERIFYPEER => false,
        ]);
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $err = curl_error($ch);
        curl_close($ch);
        if ($resp === false) return ['success' => false, 'error' => "cURL: {$err}"];
        if ($code !== 200) return ['success' => false, 'error' => "HTTP {$code}"];
        $data = json_decode($resp, true);
        if (isset($data['error'])) return ['success' => false, 'error' => $data['error']['message'] ?? ''];
        $choice = $data['choices'][0] ?? null;
        return $choice ? ['success' => true, 'data' => $choice] : ['success' => false, 'error' => '空结果'];
    }

    private function cleanUtf8($data)
    {
        if (is_string($data)) return mb_convert_encoding($data, 'UTF-8', 'UTF-8');
        if (is_array($data)) return array_map([$this, 'cleanUtf8'], $data);
        return $data;
    }
}
```

**注意事项**:
- LLM API 必须支持 Function Calling（OpenAI 协议）
- 最大轮次建议不超过 5，防止无限循环消耗 Token
- 工具执行结果必须返回 JSON 可序列化的数据
- UTF-8 清理防止非标准字符导致 JSON 编码失败

## M02：LLM API 调用封装

**适用系统**: emlog pro  
**适用场景**: 任何需要调用大语言模型的插件。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

class LLMClient
{
    private $apiUrl;
    private $apiKey;
    private $model;
    private $timeout;

    public function __construct($apiUrl, $apiKey, $model = 'deepseek-chat', $timeout = 60)
    {
        $this->apiUrl = rtrim($apiUrl, '/');
        $this->apiKey = $apiKey;
        $this->model = $model;
        $this->timeout = $timeout;
    }

    public function chat($prompt, $system = '')
    {
        $messages = [];
        if (!empty($system)) $messages[] = ['role' => 'system', 'content' => $system];
        $messages[] = ['role' => 'user', 'content' => $prompt];
        return $this->sendRequest($messages);
    }

    private function sendRequest($messages, $tools = [])
    {
        $body = [
            'model' => $this->model, 'messages' => $messages,
            'temperature' => 0.7, 'max_tokens' => 2000,
        ];
        if (!empty($tools)) {
            $body['tools'] = array_map(fn($t) => ['type' => 'function', 'function' => $t], $tools);
        }
        $json = json_encode($body, JSON_UNESCAPED_UNICODE);
        if ($json === false) {
            $body = $this->cleanUtf8($body);
            $json = json_encode($body, JSON_UNESCAPED_UNICODE);
        }
        $ch = curl_init($this->apiUrl);
        curl_setopt_array($ch, [
            CURLOPT_POST => true, CURLOPT_POSTFIELDS => $json,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json',
                'Authorization: Bearer ' . $this->apiKey],
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => $this->timeout,
            CURLOPT_SSL_VERIFYPEER => false,
        ]);
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $err = curl_error($ch);
        curl_close($ch);
        if ($resp === false) return ['success' => false, 'content' => '', 'error' => "cURL: {$err}"];
        if ($code !== 200) return ['success' => false, 'content' => '', 'error' => "HTTP {$code}"];
        $data = json_decode($resp, true);
        if (!$data) return ['success' => false, 'content' => '', 'error' => 'JSON解析失败'];
        if (isset($data['error'])) return ['success' => false, 'content' => '', 'error' => $data['error']['message'] ?? ''];
        $content = $data['choices'][0]['message']['content'] ?? '';
        return ['success' => true, 'content' => $content, 'error' => ''];
    }

    private function cleanUtf8($data)
    {
        if (is_string($data)) return mb_convert_encoding($data, 'UTF-8', 'UTF-8');
        if (is_array($data)) return array_map([$this, 'cleanUtf8'], $data);
        return $data;
    }
}
```

**使用示例**:

```php

$result = $llm->chat('用一句话介绍PHP', '你是一个技术专家');

**适用系统**: emlog pro  
**适用场景**: 需要持久化存储对话记录的插件。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

class AIAgent_ChatHistory
{
    private $db;
    private $table;

    public function __construct($tableName = 'ai_agent_chat')
    {
        $this->db = MySql::getInstance();
        $this->table = DB_PREFIX . $tableName;
    }

    public function createTable()
    {
        $this->db->query("CREATE TABLE IF NOT EXISTS `{$this->table}` (
            `id` int(11) unsigned NOT NULL AUTO_INCREMENT,
            `session_id` varchar(64) NOT NULL DEFAULT '',
            `uid` int(11) unsigned NOT NULL DEFAULT 0,
            `role` varchar(20) NOT NULL DEFAULT 'user',
            `message` text NOT NULL,
            `response` text NOT NULL,
            `tool_calls` text,
            `created_at` datetime NOT NULL,
            PRIMARY KEY (`id`),
            KEY `idx_session` (`session_id`),
            KEY `idx_uid` (`uid`),
            KEY `idx_created` (`created_at`)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;");
    }

    public function dropTable()
    {
        $this->db->query("DROP TABLE IF EXISTS `{$this->table}`");
    }

    public function save($session_id, $uid, $message, $response, $tool_calls = [])
    {
        $sid = addslashes($session_id);
        $uid = intval($uid);
        $msg = addslashes($message);
        $resp = addslashes($response);
        $meta = addslashes(json_encode($tool_calls, JSON_UNESCAPED_UNICODE));
        $now = date('Y-m-d H:i:s');
        $this->db->query("INSERT INTO `{$this->table}` (session_id,uid,role,message,response,tool_calls,created_at)
            VALUES ('{$sid}',{$uid},'user','{$msg}','{$resp}','{$meta}','{$now}')");
        return $this->db->insert_id();
    }

    public function getBySession($session_id, $limit = 50)
    {
        $sid = addslashes($session_id);
        $result = [];
        $q = $this->db->query("SELECT * FROM `{$this->table}` WHERE session_id='{$sid}' ORDER BY id ASC LIMIT " . intval($limit));
        while ($r = $this->db->fetch_array($q)) {
            $result[] = [
                'id' => intval($r['id']), 'message' => $r['message'],
                'response' => $r['response'],
                'tool_calls' => json_decode($r['tool_calls'], true) ?: [],
                'created_at' => $r['created_at'],
            ];
        }
        return $result;
    }
}

## M12：频率限制（防刷）

**适用系统**: emlog pro  
**适用场景**: 需要防止 API 被恶意刷调用的插件。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

function ai_agent_check_rate_limit($uid, $perMin = 20, $perHour = 200)
{
    $db = MySql::getInstance();
    $table = DB_PREFIX . 'ai_agent_chat';
    $idClause = $uid > 0 ? "uid={$uid}" : "uid=0";
    $minAgo = date('Y-m-d H:i:s', time() - 60);
    $r = $db->once_fetch_array("SELECT COUNT(*) AS cnt FROM {$table} WHERE {$idClause} AND created_at > '{$minAgo}'");
    if (intval($r['cnt']) >= $perMin) {
        return ['ok' => false, 'msg' => "请求过于频繁，每分钟最多{$perMin}次"];
    }
    $hourAgo = date('Y-m-d H:i:s', time() - 3600);
    $r2 = $db->once_fetch_array("SELECT COUNT(*) AS cnt FROM {$table} WHERE {$idClause} AND created_at > '{$hourAgo}'");
    if (intval($r2['cnt']) >= $perHour) {
        return ['ok' => false, 'msg' => "本小时请求已达上限（{$perHour}次）"];
    }
    return ['ok' => true, 'msg' => ''];
}

function ai_agent_content_filter($content)
{
    if (empty($content)) return $content;
    $content = preg_replace('/<script[^>]*>.*?<\/script>/is', '', $content);
    $content = preg_replace('/\s+on\w+\s*=\s*["\'][^"\'\n]*["\']/i', '', $content);
    $content = preg_replace('/javascript\s*:/i', '', $content);
    $content = preg_replace('/data\s*:\s*text\/html[^\s]*/i', '', $content);
    $content = preg_replace('/vbscript\s*:/i', '', $content);
    return $content;
}

## M17：AI 自动摘要（save_log 钩子）

**适用系统**: emlog pro  
**适用场景**: 文章保存时自动生成摘要。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

function ai_agent_auto_summary_and_tags($blogid, $pubPost, $logData)
{
    if (empty($blogid)) return;
    $config = ai_agent_get_config();
    if (empty($config['llm_api_url']) || empty($config['llm_api_key'])) return;
    $db = MySql::getInstance();
    $row = $db->once_fetch_array("SELECT gid, title, content, excerpt, tags FROM " . DB_PREFIX . "blog WHERE gid=" . intval($blogid));
    if (!$row) return;
    $plain = strip_tags($row['content']);
    if (mb_strlen($plain) < 50) return;
    $sample = mb_substr($plain, 0, 2000);
    $need_excerpt = empty($row['excerpt']);
    $need_tags = empty($row['tags']);
    if (!$need_excerpt && !$need_tags) return;

    // 构建 Prompt 并调用 LLM
    $parts = [];
    if ($need_excerpt) $parts[] = "1. Generate a summary within 150 chars";
    if ($need_tags) $parts[] = "2. Recommend 3-5 tags, comma separated";
    $sys = "You are a blog content assistant. Tasks:\n" . implode("\n", $parts) . "\n\nJSON:\n{";
    $jp = [];
    if ($need_excerpt) $jp[] = '"excerpt":"summary"';
    if ($need_tags) $jp[] = '"tags":"tag1,tag2"';
    $sys .= implode(",", $jp) . "}";

    $body = [
        'model' => $config['llm_model'],
        'messages' => [
            ['role' => 'system', 'content' => $sys],
            ['role' => 'user', 'content' => "Title: {$row['title']}\n\nContent:\n{$sample}"],
        ],
        'temperature' => 0.3, 'max_tokens' => 500,
    ];
    $ch = curl_init($config['llm_api_url']);
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => json_encode($body, JSON_UNESCAPED_UNICODE),
        CURLOPT_HTTPHEADER => ['Content-Type: application/json', 'Authorization: Bearer ' . $config['llm_api_key']],
        CURLOPT_RETURNTRANSFER => true, CURLOPT_TIMEOUT => 30, CURLOPT_SSL_VERIFYPEER => false,
    ]);
    $resp = curl_exec($ch);
    $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    if ($resp === false || $code !== 200) return;

    $data = json_decode($resp, true);
    $ai = $data['choices'][0]['message']['content'] ?? '';
    if (empty($ai)) return;
    $ai = preg_replace('/```json\s*/', '', $ai);
    $ai = preg_replace('/```\s*$/', '', $ai);
    $result = json_decode(trim($ai), true);
    if (!$result) return;

    $fields = [];
    if ($need_excerpt && !empty($result['excerpt'])) {
        $ex = addslashes(mb_substr($result['excerpt'], 0, 300));
        $fields[] = "excerpt='{$ex}'";
    }
    if ($need_tags && !empty($result['tags'])) {
        $tag_model = new Tag_Model();
        $tag_model->updateTag($result['tags'], $blogid);
    }
    if (!empty($fields)) {
        $db->query("UPDATE " . DB_PREFIX . "blog SET " . implode(",", $fields) . " WHERE gid=" . intval($blogid));
    }
}
addAction('save_log', 'ai_agent_auto_summary_and_tags');
```

**注意事项**:
- 仅在 excerpt 字段为空时生成摘要
- 仅在 tags 字段为空时推荐标签
- 文章内容少于50字不触发
- LLM调用失败不影响文章正常保存

function ai_agent_get_config()
{
    static $_cache = null;
    if ($_cache !== null) {
        return $_cache;
    }
    $storage = Storage::getInstance('ai_agent');
    $stored = $storage->getValue('config');
    $defaults = [ /* ... 默认配置 ... */ ];
    if (empty($stored)) {
        $config = $defaults;
    } else {
        $config = array_merge($defaults, $stored);

## M23：图片 MIME 验证

**适用系统**: emlog pro  
**适用场景**: 需要验证下载文件类型的插件。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

function downloadAndSaveImage($remote_url, $save_path)
{
    $ch = curl_init($remote_url);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true, CURLOPT_TIMEOUT => 30,
        CURLOPT_SSL_VERIFYPEER => false, CURLOPT_FOLLOWLOCATION => true,
    ]);
    $image_data = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    if ($image_data === false || $http_code !== 200) return '';

    // MIME 验证
    if (function_exists('finfo_open')) {
        $finfo = finfo_open(FILEINFO_MIME_TYPE);
        $mime = finfo_buffer($finfo, $image_data);
        finfo_close($finfo);
        $allowed = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
        if (!in_array($mime, $allowed)) return '';
    }

    return file_put_contents($save_path, $image_data) !== false ? $save_path : '';
}
```

## M24：缓存目录可写性检查

**适用系统**: emlog pro  
**适用场景**: 插件启用时检查 emlog 缓存目录是否可写，避免因权限问题导致静默失败。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

function ai_agent_check_cache_writable()
{
    $cache_dir = EMLOG_ROOT . '/content/cache';
    if (!is_dir($cache_dir) || !is_writable($cache_dir)) {
        return [
            'ok' => false,
            'msg' => 'emlog 缓存目录 content/cache 不可写，请检查目录权限（建议 755 或 777）',
        ];
    }
    return ['ok' => true, 'msg' => ''];
}
```

**使用示例**:

```php
// 在 callback_init() 开头调用
function callback_init()
{
    $check = ai_agent_check_cache_writable();
    if (!$check['ok']) {
        emMsg('插件启用失败：' . $check['msg']);
        return;
    }
    // ... 初始化逻辑
}
```

**注意事项**:
- emlog 启用插件后会调用 `$CACHE->updateCache('options')` 写入 `content/cache`
- 目录不可写时缓存更新失败，但不会报错（静默失败）
- 使用 `emMsg()` 输出友好提示，用户可自行修复权限

## M25：表单 AJAX 异步提交框架

**适用系统**: emlog pro  
**适用场景**: 插件后台设置页需要无刷新保存表单数据。

**核心代码**:

```php
<!-- HTML 表单结构 -->
<form method="post" id="my_plugin_form" class="ai-agent-form"
      action="./plugin.php?plugin=my_plugin&action=save_setting">
    <input type="hidden" name="ai_agent_action" value="save_basic">
    <input type="text" name="field1" value="">
    <button type="submit">保存</button>
</form>

<script>
// emlog 内置 submitForm 函数（位于 admin/views/js/common.js）
// 自动拦截 .ai-agent-form 的 submit 事件
$(".ai-agent-form").submit(function(event) {
    event.preventDefault();
    submitForm("#" + $(this).attr("id"));
});

// submitForm 实现：
function submitForm(formId, successMsg) {
    $.ajax({
        type: "POST",
        url: $(formId).attr('action'),
        data: $(formId).serialize(),
        success: function () {
            cocoMessage.success(successMsg || '保存成功')
        },
        error: function (xhr) {
            const errorMsg = JSON.parse(xhr.responseText).msg;
            cocoMessage.error(errorMsg, 4000)
        }
    });
}
</script>
```

**使用示例**:

```php
// 后端处理（plugin_setting 函数）
function plugin_setting()
{
    $ai_action = Input::postStrVar('ai_agent_action', '');
    switch ($ai_action) {
        case 'save_basic':
            // 处理基础设置保存
            $config = ai_agent_get_config();
            $config['field1'] = Input::postStrVar('field1', '');
            ai_agent_save_config($config);
            Output::ok();
            break;
        case 'save_other':
            // 处理其他保存
            break;
    }
}
```

**注意事项**:
- ⚠️ 表单 action 必须是 `action=save_setting`（emlog 后台只处理此路由）
- 通过 `ai_agent_action` 隐藏字段区分不同表单的操作
- emlog 后台 `plugin.php` 不处理 `action=save_basic` 等自定义 action（会静默忽略）
- `submitForm` 的 success 回调不检查响应内容，HTTP 200 就显示"保存成功"

## M26：配置静态缓存 + array_merge 默认值合并

**适用系统**: emlog pro  
**适用场景**: 需要频繁读取配置的插件，避免重复查询数据库。

**核心代码**:

```php
<?php
!defined('EMLOG_ROOT') && exit('access denied!');

function ai_agent_get_config()
{
    // 静态缓存：同一请求内只读一次数据库
    static $cache = null;
    if ($cache !== null) return $cache;

    $storage = Storage::getInstance('ai_agent');
    $stored = $storage->getValue('config');

    // 默认配置（新字段在此添加，旧安装自动获得）
    $defaults = [
        'field1' => 'default_value',
        'field2' => 0,
        'version' => '1.0.0',
    ];

    // 合并：默认值 + 已存储值（已存储值优先）
    if (empty($stored)) {
        $config = $defaults;
    } else {
        $config = array_merge($defaults, $stored);
    }

    $cache = $config;
    return $config;
}

function ai_agent_save_config($config)
{
    $storage = Storage::getInstance('ai_agent');
    $config['version'] = AI_AGENT_VERSION;
    $storage->setValue('config', $config, 'array');
    // 注意：保存后同一请求内静态缓存仍为旧值
    // 下次请求时会自动读取新值
}
```

**使用示例**:

```php
// 读取配置（自动缓存）
$config = ai_agent_get_config();
echo $config['field1'];

// 保存配置
$config = ai_agent_get_config();
$config['field1'] = 'new_value';
ai_agent_save_config($config);
```

**注意事项**:
- `array_merge` 确保旧安装也能获得新字段（默认值填充）
- 静态缓存在同一请求内有效，避免重复读取 Storage
- 保存后缓存不会立即更新（下一次请求才生效）
- 如果需要在同一请求中立即看到更新，需手动重置缓存（`$cache = null`）

## M27：emlog 插件 callback 自包含依赖加载模式

**适用系统**: emlog pro  
**适用场景**: 插件的 callback 文件需要使用主文件中的函数或类。

**问题背景**:

emlog 核心 `Plugin_Model::activePlugin()` 启用插件时只加载 `callback.php`，不加载主文件 `plugin.php`。如果 `callback_init()` 依赖主文件中的函数或类，会导致 Fatal Error。

**核心代码**:

```php
<?php
/**
 * 插件 callback 文件
 * emlog 启用插件时只加载此文件，不加载主文件
 */
defined('EMLOG_ROOT') || exit('access denied!');

// 关键：手动加载主插件文件，确保所有依赖可用
// 使用 require_once 避免重复加载（正常运行时主文件已由 init.php 加载）
require_once __DIR__ . '/my_plugin.php';

function callback_init()
{
    // 现在可以安全使用主文件中的函数和类
    $config = my_plugin_get_config();
    $history = new MyPlugin_ChatHistory();
    $history->createTable();
}

function callback_rm()
{
    $storage = Storage::getInstance('my_plugin');
    $storage->deleteAllName('YES');
    $history = new MyPlugin_ChatHistory();
    $history->dropTable();
}

function callback_up()
{
    $storage = Storage::getInstance('my_plugin');
    $config = $storage->getValue('config');
    if (!empty($config)) {
        $defaults = my_plugin_get_config();
        $config = array_merge($defaults, $config);
        $config['version'] = MY_PLUGIN_VERSION;
        $storage->setValue('config', $config, 'array');
    }
}
```

**使用示例**:

```php
// 主文件 my_plugin.php
<?php
defined('EMLOG_ROOT') || exit('access denied!');

define('MY_PLUGIN_VERSION', '1.0.0');
define('MY_PLUGIN_DIR', __DIR__);

require_once MY_PLUGIN_DIR . '/lib/SomeClass.php';

function my_plugin_get_config() { /* ... */ }

// callback 文件 my_plugin_callback.php
<?php
defined('EMLOG_ROOT') || exit('access denied!');

// 必须手动加载主文件！
require_once __DIR__ . '/my_plugin.php';

function callback_init() { /* 安全使用所有依赖 */ }
```

**注意事项**:
- ⚠️ **必须**在 callback 文件顶部 `require_once` 主文件
- `require_once` 安全可重复加载（正常运行时主文件已由 `init.php` 加载，不会重复定义）
- 不要依赖 `init.php` 的加载顺序（启用/禁用时主文件未被加载）
- 如果主文件较大，可以只加载所需的子文件（如 `require_once __DIR__ . '/lib/ChatHistory.php'`）
- 删除 callback 文件中的 `require_once` 前，确认所有依赖都已自包含
